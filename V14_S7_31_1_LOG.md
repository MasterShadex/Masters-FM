```
private const string PinnedThumbprint   = "4B1660FC0B77F55C7D47B3B9010C873E5CC2B2BF";
private const string PinnedPublicKeyHex = "3082...010203010001";  // full 540 hex

private bool VerifyAuthenticode(string msiPath, out string error):
    error = ""
    try:
        signer = new X509Certificate2(X509Certificate.CreateFromSignedFile(msiPath))  // unsigned -> throws -> reject
        if !signer.Thumbprint.Equals(PinnedThumbprint, OrdinalIgnoreCase):  error="thumbprint mismatch"; return false
        if !signer.GetPublicKeyString().Equals(PinnedPublicKeyHex, OrdinalIgnoreCase): error="pubkey mismatch"; return false
        if !signer.Subject.Contains("CN="+ExpectedCertCn, Ordinal): error="subject mismatch"; return false
        return true                       // pinned to OUR cert; NO chain.Build, NO Trusted-Root dependency
    catch (ex): error="exception: "+ex.Message; return false
```
