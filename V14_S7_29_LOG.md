## S0.5 Theme apply algorithm LOCK

```javascript
function applyTheme(themeKey) {
    const theme = THEMES[themeKey];
    if (!theme) return;

    // For each mapped nested path, write into State.config[flat-id].
    for (const [nestedPath, flatId] of Object.entries(THEME_KEY_MAP)) {
        const value = getNestedValue(theme, nestedPath);
        if (value === undefined) continue;
        // border.colors is an array -> comma-separated for colorList input
        const writable = Array.isArray(value) ? value.join(',') : value;
        State.config[flatId] = writable;
    }
    State.config['c-theme'] = themeKey;  // remember selection
    State.isDirty = true;

    // Sentinel substitution is automatic: the literal string 'var(--accent-master)'
    // gets written to the per-element CSS var by applyControlValue, and the
    // overlay CSS resolves the var() at runtime. So just setting the value
    // is enough -- no manual substitution.

    refreshControlValues();
    applyConfig();
    refreshAllMasterBadges();
    previewConfig();
}

function getNestedValue(obj, path) {
    return path.split('.').reduce((o, k) => (o == null ? undefined : o[k]), obj);
}
```

Empty-theme handling: `'Rainbow (Default)'` is `{}` -- the loop does nothing,
which means current State.config stays. Per current customize.html the
'Rainbow (Default)' theme acts as "DEFAULTS" -- but our Reset already does
that via onResetDefaults. Acceptable behaviour: picking 'Rainbow (Default)'
doesn't reset; user clicks Reset for a clean wipe.

```javascript
const MASTER_FOLLOWING_IDS = [
    'c-np-color', 'c-bars-color', 'c-platform-color', 'c-platform-dot',

function updateMasterBadge(id) {
    const row = document.querySelector(`[data-control-id="${id}"]`);
    if (!row) return;
    const val = State.config[id];
    const isFollowing = val === 'var(--accent-master)';
    let badge = row.querySelector('.master-badge');
    if (isFollowing) {
        if (!badge) {
            badge = document.createElement('span');
            badge.className = 'master-badge';
            badge.textContent = 'Following master';
            badge.title = 'Click to set independent color';
            badge.addEventListener('click', () => unfollowMaster(id));
            const grp = row.querySelector('.control-input-group');
            if (grp) grp.appendChild(badge);
        }
    } else if (badge) badge.remove();
}

function unfollowMaster(id) {
    const resolved = (getComputedStyle(document.documentElement)
        .getPropertyValue('--accent-master').trim() || '#c060ff');
    State.config[id] = resolved;
    State.isDirty = true;
    refreshControlValues();
    applyConfig();
    updateMasterBadge(id);
    previewConfig();
}

function refreshAllMasterBadges() {
    for (const id of MASTER_FOLLOWING_IDS) updateMasterBadge(id);
}
```

## S0.7 Search filter design LOCK

```javascript
function filterControls(query) {
    const q = (query || '').trim().toLowerCase();
    const rows = document.querySelectorAll('.control-row');
    const supercats = document.querySelectorAll('.supercat');

    if (!q) {
        rows.forEach(r => { r.hidden = false; });
        supercats.forEach(sc => { sc.hidden = false; });
        return;
    }

    rows.forEach(row => {
        const id = (row.dataset.controlId || '').toLowerCase();
        const label = (row.querySelector('.control-label')?.textContent || '').toLowerCase();
        // JARGON_MAP key = label; value = space-separated synonyms
        const jargon = JARGON_MAP[label] || '';
        const match = label.includes(q) || id.includes(q) || (jargon && jargon.toLowerCase().includes(q));
        row.hidden = !match;
    });

    // Hide a supercat entirely if all its rows are hidden.
    supercats.forEach(sc => {
        const visible = sc.querySelectorAll('.control-row:not([hidden])').length;
        sc.hidden = visible === 0;
    });
}
```

Wiring:
- `#settings-search` `input` event -> `filterControls(value)` + toggle clear button hidden
- `#settings-search-clear` `click` -> clear input + filterControls('') + focus input
- Document `keydown` Ctrl+F -> preventDefault + focus + select

```javascript
function initSupercatCollapse() {
    document.querySelectorAll('.supercat-header').forEach(header => {
        const key = header.dataset.supercat;
        if (!key) return;
        const supercat = header.closest('.supercat');
        if (!supercat) return;
        const lsKey = `supercat_${key}_collapsed`;

        // Restore from localStorage. '1' -> collapsed; '0' -> expanded; null
        // -> default (start expanded, others expanded too per current behaviour).
        let stored = null;
        try { stored = localStorage.getItem(lsKey); } catch {}
        if (stored === '1') supercat.classList.add('collapsed');
        // First-load (null) keeps 'start' expanded explicitly; others stay
        // expanded by default behaviour (no class added).

        header.addEventListener('click', () => {
            supercat.classList.toggle('collapsed');
            const collapsed = supercat.classList.contains('collapsed');
            try { localStorage.setItem(lsKey, collapsed ? '1' : '0'); } catch {}
        });
    });
}
```
